# A VERSAO DO JOGO em cada linha de tentativas.csv (Tentativas.h): o commit de
# onde o executavel saiu, para a analise separar as tentativas de antes e de
# depois de uma mudanca de balanceamento. Sem git (um zip do codigo), "dev".
execute_process(COMMAND git rev-parse --short HEAD
        WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
        OUTPUT_VARIABLE DPI_VERSAO OUTPUT_STRIP_TRAILING_WHITESPACE
        RESULT_VARIABLE DPI_VERSAO_ERRO ERROR_QUIET)
if(NOT DPI_VERSAO_ERRO EQUAL 0 OR DPI_VERSAO STREQUAL "")
    set(DPI_VERSAO "dev")
endif()
target_compile_definitions(${PROJECT_NAME} PRIVATE DPI_VERSAO="${DPI_VERSAO}")

target_link_libraries(${PROJECT_NAME} ${SDL2_LIBRARIES} SDL2_image::SDL2_image SDL2_mixer::SDL2_mixer SDL2_ttf::SDL2_ttf)

# NAO adicione Source/Core ao include path deste alvo (nem de nenhum outro).
# O projeto tem Source/Core/Math.h, e no Windows (e no macOS) o sistema de
# arquivos nao diferencia maiusculas: com Source/Core no include path, o
# #include <math.h> da PROPRIA biblioteca padrao encontra Source/Core/Math.h em
# vez do cabecalho C, e sinf/fabs/sqrtf deixam de existir - centenas de
# erros em cmath e cstdlib. Os arquivos deste alvo ja usam includes
# relativos, entao nao precisam do caminho.

# Faz o doctest incluir os cabecalhos reais da biblioteca padrao em vez de
# declarar std::tuple por conta propria. O MSVC recente avisa (C5285) que
# declarar templates da std fora dela e proibido pelo padrao.
target_compile_definitions(dpi_tests PRIVATE DOCTEST_CONFIG_USE_STD_HEADERS)

# Caminho absoluto de Assets/, para tests/Data/test_arquivos_de_dados.cpp poder abrir
# os arquivos que o JOGO carrega e conferir que eles estao corretos.
#
# Vem daqui, e nao de um caminho relativo, porque o diretorio de trabalho do
# executavel de teste depende de quem o chama: o CLion roda de um lugar, o ctest
# de outro e o terminal de um terceiro. Um "../Assets" funcionaria num deles e
# faria o teste falhar por motivo errado nos outros dois.
target_compile_definitions(dpi_tests PRIVATE DPI_ASSETS_DIR="${CMAKE_CURRENT_SOURCE_DIR}/Assets")

add_test(NAME unidade COMMAND dpi_tests)
